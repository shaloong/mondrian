cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(mondrian_reference_aja LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This qualified bridge build currently targets Windows only")
endif()
if(NOT EXISTS "${MONDRIAN_AJA_SDK}/ajantv2/includes/ntv2card.h")
  message(FATAL_ERROR "MONDRIAN_AJA_SDK must point to the pinned libajantv2 checkout")
endif()
set(AJANTV2_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(AJANTV2_DISABLE_DRIVER ON CACHE BOOL "" FORCE)
set(AJANTV2_DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(AJANTV2_DISABLE_TOOLS ON CACHE BOOL "" FORCE)
set(AJANTV2_DISABLE_PLUGIN_LOAD ON CACHE BOOL "" FORCE)
set(AJANTV2_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(AJA_MSVC_RUNTIME_LIBRARY MultiThreaded CACHE STRING "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_subdirectory("${MONDRIAN_AJA_SDK}" sdk EXCLUDE_FROM_ALL)
add_library(mondrian_reference_aja SHARED bridge.cpp vanc_raster.cpp)
target_compile_features(mondrian_reference_aja PRIVATE cxx_std_17)
target_compile_definitions(mondrian_reference_aja PRIVATE MD_AJA_BUILD NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(mondrian_reference_aja PRIVATE ajantv2)
target_compile_options(mondrian_reference_aja PRIVATE /W4 /EHsc)
add_executable(mondrian_aja_abi_probe abi_probe.cpp)
target_compile_features(mondrian_aja_abi_probe PRIVATE cxx_std_17)
target_link_libraries(mondrian_aja_abi_probe PRIVATE mondrian_reference_aja)
target_compile_options(mondrian_aja_abi_probe PRIVATE /W4 /WX /EHsc)
add_executable(mondrian_aja_vanc_probe vanc_probe.cpp vanc_raster.cpp)
target_compile_features(mondrian_aja_vanc_probe PRIVATE cxx_std_17)
target_compile_options(mondrian_aja_vanc_probe PRIVATE /W4 /WX /EHsc)
